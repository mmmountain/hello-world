function parseCSV(text) {

  const rows = []; let row = [], f = "", q = false;

  for (let i = 0; i < text.length; i++) {

    const c = text[i];

    if (q) {

      if (c === '"') { if (text[i + 1] === '"') { f += '"'; i++; } else q = false; }

      else f += c;

    } else if (c === '"') q = true;

    else if (c === ",") { row.push(f); f = ""; }

    else if (c === "\n" || c === "\r") {

      if (c === "\r" && text[i + 1] === "\n") i++;

      row.push(f); rows.push(row); row = []; f = "";

    } else f += c;

  }

  if (f || row.length) { row.push(f); rows.push(row); }

  return rows;

}

function at(d, hhmm) { const [y, m, dd] = DAYS[d].date; const [h, mi] = hhmm.split(":").map(Number); return new Date(y, m, dd, h, mi); }

function fmt(hhmm) {

  let [h, m] = hhmm.split(":").map(Number); const ap = h >= 12 ? "pm" : "am"; h = h % 12 || 12;

  return m ? `${h}:${String(m).padStart(2, "0")}` + ap : `${h}` + ap;

}

function range(s) {

  const a = fmt(s.start), b = fmt(s.end);

  const same = a.slice(-2) === b.slice(-2);

  return (same ? a.slice(0, -2) : a) + "–" + b;

}

function esc(t) { return String(t).replace(/[&<>"]/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" })[c]); }

function status(s, now) { const a = at(s.day, s.start), b = at(s.day, s.end); return now >= b ? "done" : now >= a ? "now" : "soon"; }

function matches(s) {

  if (!query) return true;

  const hay = [s.who, s.title, s.label, s.room, ...(s.desc || []), ...(s.bio || [])].join(" ").toLowerCase();

  return query.toLowerCase().split(/\s+/).every(w => hay.includes(w));

}


function groupSlots(list) {

  const map = new Map();

  for (const s of list) {

    const k = s.day + s.start + s.end;

    if (!map.has(k)) map.set(k, { day: s.day, start: s.start, end: s.end, items: [] });

    map.get(k).items.push(s);

  }

  const rank = { class: 0, performance: 0, circle: 1, jam: 2, break: 3, logistics: 4 };

  return [...map.values()]

    .sort((a, b) => ORDER.indexOf(a.day) - ORDER.indexOf(b.day) || a.start.localeCompare(b.start) || a.end.localeCompare(b.end))

    .map(g => (g.items.sort((x, y) => rank[x.kind] - rank[y.kind]), g));

}


function cardHTML(s) {

  const isClass = s.kind === "class";

  const hasMore = (s.desc && s.desc.length) || (s.bio && s.bio.length) || s.link || s.kind === "jam";

  const open = openIds.has(s.id);

  const quiet = s.kind === "break" || s.kind === "logistics";

  const tag = isClass ? "Class" : s.kind === "jam" ? "Jam" : s.kind === "performance" ? "Show" : "";

  let head = isClass

    ? `<span class="who">${esc(s.who)}</span><span class="ctitle">${esc(s.title || "Class description not posted yet")}</span>`

    : `<span class="ctitle">${esc(s.title)}</span>`;

  const where = `<span class="where">${tag ? `<span class="tag">${tag}</span>` : ""}${s.room ? esc(s.room) : ""}${query ? ` · ${DAYS[s.day].label.slice(0, 3)} ${range(s)}` : ""}</span>`;

  const more = hasMore && !open ? `<span class="more">${isClass ? "Read description" : "Details"}</span>` : "";

  let detail = "";

  if (open && hasMore) {

    detail = `<div class="detail" id="d-${s.id}">`;

    for (const p of s.desc || []) detail += `<p>${esc(p)}</p>`;

    if (s.bio && s.bio.length) { detail += `<h4>About the teacher${s.who.includes("&") || s.who.includes("Collective") ? "s" : ""}</h4>`; for (const p of s.bio) detail += `<p class="bio">${esc(p)}</p>`; }

    if (s.kind === "jam") detail += `<button class="linkbtn" data-agree>Read the community agreements</button>`;

    if (s.link) detail += `<a href="${esc(s.link)}" target="_blank" rel="noopener">Map to Harrison Theatre ↗</a>`;

    detail += `</div>`;

  }

  const m = marks.get(s.id) || 0;

  const markBtn = isClass || s.kind === "performance" || s.kind === "jam"

    ? `<button class="markbtn" data-mark-id="${s.id}" data-mark="${m}" aria-label="${["Mark as interested", "Interested. Tap to mark as going", "Going. Tap to clear"][m]}">${[STAR_OFF, STAR_ICON, CHECK_ICON][m]}<small>${["Mark", "Maybe", "Going"][m]}</small></button>` : "";

  const crewLine = crewHTML(s.id);

  return `<article class="card kind-${s.kind}${quiet ? " quiet" : ""}">

    <div class="row">

      <button class="opener" data-open="${s.id}" ${hasMore ? `aria-expanded="${open}"` : "disabled"}>${head}${where}${more}</button>

      ${markBtn}

    </div>${crewLine}${detail}</article>`;

}


function whoMarked(id) {

  const go = [], maybe = [];

  const mine = marks.get(id);

  if (mine === 2) go.push("You"); else if (mine === 1) maybe.push("You");

  for (const [uid, p] of Object.entries(crew)) {

    if (uid === myId) continue;

    const v = p.marks && p.marks[id];

    if (v === 2) go.push(p.name); else if (v === 1) maybe.push(p.name);

  }

  return { go, maybe };

}

function crewHTML(id) {

  const { go, maybe } = whoMarked(id);

  const others = go.concat(maybe).filter(n => n !== "You");

  if (!others.length) return "";

  return `<div class="crew">${go.length ? `<span class="go"><b>Going</b>${go.map(esc).join(", ")}</span>` : ""}${maybe.length ? `<span class="maybe"><b>Interested</b>${maybe.map(esc).join(", ")}</span>` : ""}</div>`;

}

function othersCount() { return Object.keys(crew).filter(uid => uid !== myId).length; }

function crewViewHTML(now) {

  if (people.length) {

    html += `<div class="roster">${people.sort((a, b) => a[1].name.localeCompare(b[1].name)).map(([, p]) => {

      const vals = Object.values(p.marks || {});

      return `<span>${esc(p.name)} <small>${vals.filter(v => v === 2).length} going · ${vals.filter(v => v === 1).length} maybe</small></span>`;

}

}

function nowCardHTML(now) {

  const first = at("thu", "17:30"), last = at("sat", "22:30");

  if (now < first) {

    const days = Math.ceil((new Date(2026, 9, 15) - new Date(now.getFullYear(), now.getMonth(), now.getDate())) / 864e5);

    return `<div class="nowcard"><span class="eyebrow">Coming up</span><p class="big">Registration opens Thursday at 5:30pm${days === 1 ? " — tomorrow" : days > 1 ? ` — in ${days} days` : ""}.</p><p>Tap the star on a class once for Interested, twice for Going. Your crew sees your marks.</p></div>`;

  }

  if (now > last) return "";

  const all = DATA.sessions.filter(s => s.kind !== "logistics");

  const live = all.filter(s => status(s, now) === "now" && s.kind !== "break");

  const upcoming = all.filter(s => at(s.day, s.start) > now && s.kind !== "break").sort((a, b) => at(a.day, a.start) - at(b.day, b.start));

  const nextStart = upcoming[0];

  const next = nextStart ? upcoming.filter(s => s.day === nextStart.day && s.start === nextStart.start) : [];

  const pickNext = next.filter(s => marks.has(s.id));

  let html = `<div class="nowcard"><span class="eyebrow">Right now</span>`;

  html += live.length ? `<p class="big">${live.map(s => esc(s.kind === "class" ? s.who : s.title)).join(" · ")}</p>` : `<p class="big">Break</p>`;

  if (nextStart) {

    const mins = Math.round((at(nextStart.day, nextStart.start) - now) / 6e4);

    const when = mins < 90 ? `in ${mins} min` : `${DAYS[nextStart.day].label} ${fmt(nextStart.start)}`;

    html += `<p><b>Next, ${when}:</b> ${pickNext.length ? "★ " + pickNext.map(s => esc((s.who || s.title) + " in " + s.room)).join(", ") : next.length + (next.length === 1 ? " session" : " sessions") + " — " + next.map(s => esc(s.who || s.title)).join(", ")}</p>`;

  }

  return html + `</div>`;

}


function render() {

  document.getElementById("crewcount").textContent = share === "on" || (share === "off" && gotFirstSnap) ? (n + 1) + " people" : "—";

  let html = meBarHTML();

  if (query) {

    const hits = DATA.sessions.filter(s => s.kind !== "break" && s.kind !== "logistics" && matches(s));

    html += hits.length

      ? `<div class="list">${hits.map(cardHTML).join("")}</div>`

      : `<p class="empty">Nothing matches “${esc(query)}”.</p>`;

  } else if (day === "crew") {

    html += crewViewHTML(now);

  } else if (day === "picks") {

    html += nowCardHTML(now);

    const picks = DATA.sessions.filter(s => marks.has(s.id));

    if (!picks.length) html += `<p class="empty">Tap the star on a class once for Interested, twice for Going.<br>Your weekend shows up here, with time clashes flagged.</p>`;

    for (const d of ORDER) {

      const mine = picks.filter(s => s.day === d);

      if (!mine.length) continue;

      const slots = groupSlots(mine);

      html += `<section class="slot"><div class="slot-head"><h2 class="slot-time">${DAYS[d].label}</h2></div><div class="list">`;

      for (const g of slots) {

        const clash = g.items.length > 1 ? `<span class="conflict">${g.items.length} picks at ${range(g)}. Choose one.</span>` : "";

        html += `<div class="slot"><div class="slot-head"><span class="slot-meta">${range(g)}</span>${clash}</div>${g.items.map(cardHTML).join("")}</div>`;

      }

      html += `</div></section>`;

    }

  } else {

    html += nowCardHTML(now);

    html += slots.map(g => slotHTML(g, now, false)).join("");

}

}

const sheetReady = /^https:\/\/script\.google\.com\/.+\/exec$/.test(SCRIPT_URL);

let writing = false, pending = false, saveTimer = null, dirty = false;

function saveMine() {

  store.set("tdif26-marks", Object.fromEntries(marks));

  if (!sheetReady || !myName) return;

  dirty = true;

  crew[myId] = { name: myName, marks: Object.fromEntries(marks) };

  clearTimeout(saveTimer);

  saveTimer = setTimeout(flush, 600);

}

async function flush() {

  if (writing) { pending = true; return; }

  writing = true; dirty = false;

  try {

    const res = await fetch(SCRIPT_URL, {

      method: "POST",

      headers: { "Content-Type": "text/plain;charset=utf-8" },

      body: JSON.stringify({ id: myId, name: myName, marks: Object.fromEntries(marks) })

    });

    if (!res.ok) throw new Error(res.status);

  } catch (e) {

    dirty = true; share = "off"; render();

  }

  writing = false;

  if (pending) { pending = false; flush(); }

}

function cleanMarks(obj) {

  const out = {};

  if (obj && typeof obj === "object") for (const [k, v] of Object.entries(obj)) if (v === 1 || v === 2) out[String(k)] = v;

  return out;

}

async function pull() {

  if (!sheetReady) return;

  try {

    const res = await fetch(SCRIPT_URL + "?t=" + Date.now(), { cache: "no-store" });

    const j = await res.json();

    const next = {};

    for (const [id, p] of Object.entries((j && j.people) || {})) {

      next[id] = { name: String((p && p.name) || "Friend").slice(0, 24), marks: cleanMarks(p && p.marks) };

    }

    if (myName) next[myId] = { name: myName, marks: Object.fromEntries(marks) };

    crew = next;

    const wasOff = share !== "on";

    share = "on"; gotFirstSnap = true;

    if (dirty || (wasOff && myName && marks.size)) flush();

    render();

  } catch (e) {

    share = "off"; render();

  }

}

function connectCrew() {

  if (!sheetReady) { share = "readonly"; render(); return; }

  pull();

}
